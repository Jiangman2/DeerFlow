本包用于复测诊断，不是生产热更包。
将 portal-glm-acceptance-20261014-r3.tar.gz 上传到 datax-01:/home/cnooc。
以下命令保留 70 个 Gateway，不重启、不写配置、不导入镜像、不执行数据库迁移。
会创建少量独立共享测试文件，并在隐藏提示中输入 GLM Key 后最多调用模型四次。
GLM 每次 max_tokens=1000、120 秒总时限；此参数不保证限制隐藏思考长度。
不要在聊天、命令参数或脚本中填写 Key。没有 Key 可去掉 --glm。

sudo bash -se <<'BASH'
set -Eeuo pipefail
umask 077
test "$(hostname -s)" = datax-01
cd /home/cnooc
PACKAGE=portal-glm-acceptance-20261014-r3.tar.gz
test -f "$PACKAGE" && test ! -L "$PACKAGE"
CONTROL=$(mktemp -d /run/portal-acceptance-r3-XXXXXXXX)
install -o root -g root -m 0600 "$PACKAGE" "$CONTROL/package.tar.gz"
test "$(sha256sum "$CONTROL/package.tar.gz" | awk '{print $1}')" = \
  ac7d3845d21dc5bdf7a215c19a305d9f0a6e797bf94925c1da431df0ef6f5e13
tar --no-same-owner --no-same-permissions -xzf "$CONTROL/package.tar.gz" -C "$CONTROL"
cd "$CONTROL"
sha256sum -c PACKAGE.SHA256
bash ./run-datax01-acceptance.sh --glm --shared-fixture
BASH
